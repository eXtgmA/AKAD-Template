% 1) Autorenformat → „Nachname V." (Nachname + Initiale mit Punkt)
\DeclareNameFormat{last-init}{%
  \namepartfamily
  \ifblank{\namepartgiveni}{}{\addspace\namepartgiveni}%
  \ifthenelse{\value{listcount}<\value{liststop}}
    {\addcomma\addspace}{}%
}
\DeclareNameAlias{author}{last-init}
\DeclareNameAlias{editor}{last-init}

% 2) Doppelpunkt nach Autorenname (in Zitationen UND Literaturverzeichnis)
%    nametitledelim wird in allen Kontexten auf Doppelpunkt gesetzt
\DeclareDelimFormat{nametitledelim}{\addcolon\addspace}
\DeclareDelimFormat[bib,biblist]{nametitledelim}{\addcolon\addspace}

% 3) Kein Verlag – nur Ort + Jahr: „Heidelberg 2002"
\renewbibmacro*{publisher+location+date}{%
  \printlist{location}%
  \iflistundef{location}{}{\setunit{\addspace}}%
  \usebibmacro{date}%
  \newunit
}

% 4) URL-Feld: „URL: <url>"
\DeclareFieldFormat{url}{URL\addcolon\addspace\url{#1}}

% 5) urldate: „[aufgerufen TT.MM.JJJJ]"
\DeclareFieldFormat{urldate}{%
  \mkbibbrackets{aufgerufen\addspace
    \iffieldundef{urlday}{}{\mkdatezeros{\thefield{urlday}}\adddot}%
    \iffieldundef{urlmonth}{}{\mkdatezeros{\thefield{urlmonth}}\adddot}%
    \thefield{urlyear}}%
}

% 7) In Zitationen: ISBN, ISSN, DOI und Auflage unterdrücken
\AtEveryCitekey{%
  \clearfield{isbn}%
  \clearfield{issn}%

  \clearfield{edition}%

        \global\firstaitoolfalse

    \fi
